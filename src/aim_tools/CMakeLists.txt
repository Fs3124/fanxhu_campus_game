cmake_minimum_required(VERSION 3.16)
project(aim_tools)
set(BUILD_TESTING OFF)
find_package(ament_cmake REQUIRED)
find_package(OpenCV 4.7.0 EXACT REQUIRED COMPONENTS core imgproc calib3d dnn)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_library(aim_tools src/armor.cpp src/small_rune.cpp src/big_rune.cpp)
target_compile_features(aim_tools PUBLIC cxx_std_17)
target_compile_options(aim_tools PRIVATE -Wall -Wextra)
target_include_directories(aim_tools PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>
  $<INSTALL_INTERFACE:include/${PROJECT_NAME}> ${OpenCV_INCLUDE_DIRS})
target_link_libraries(aim_tools PUBLIC ${OpenCV_LIBS} OpenSSL::Crypto)
install(TARGETS aim_tools EXPORT aim_tools_targets DESTINATION lib)
install(FILES src/types.hpp src/small_rune.hpp src/big_rune.hpp DESTINATION include/${PROJECT_NAME})
install(DIRECTORY models/ DESTINATION share/${PROJECT_NAME}/models)
ament_export_targets(aim_tools_targets HAS_LIBRARY_TARGET)
ament_export_libraries(aim_tools)
ament_export_dependencies(OpenCV OpenSSL)
ament_package()
