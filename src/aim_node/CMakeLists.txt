cmake_minimum_required(VERSION 3.16)
project(aim_node)
set(BUILD_TESTING OFF)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(tdt_interface REQUIRED)
find_package(aim_tools REQUIRED)
find_package(OpenCV 4.7.0 EXACT REQUIRED COMPONENTS core imgproc highgui imgcodecs)
add_executable(aim_node src/main.cpp)
target_compile_features(aim_node PRIVATE cxx_std_17)
target_compile_options(aim_node PRIVATE -Wall -Wextra)
ament_target_dependencies(aim_node rclcpp ament_index_cpp tdt_interface aim_tools)
target_link_libraries(aim_node ${OpenCV_LIBS})
install(TARGETS aim_node DESTINATION lib/${PROJECT_NAME})
ament_package()
